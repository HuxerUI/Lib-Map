cmake_minimum_required(VERSION 3.20)
project(lib_map VERSION 0.1.0 LANGUAGES NONE)

option(HUXERUI_MAP_BUILD_VIEW "Build the HuxerUI native map components" ON)
option(HUXERUI_MAP_BUILD_TESTS "Build map tests" OFF)
option(HUXERUI_MAP_BUILD_EXAMPLES "Build the map preview app" OFF)
option(HUXERUI_MAP_ENABLE_AMAP "Enable the iOS AMap backend (Android includes both providers)" ON)
option(HUXERUI_MAP_ENABLE_BAIDU "Enable the iOS Baidu backend (Android includes both providers)" ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/HuxerUIProject.cmake")
if(HUXERUI_PROJECT_PLAN_ONLY)
    return()
endif()

set(_map_sources src/types.cpp)
if(HUXERUI_MAP_BUILD_VIEW)
    list(APPEND _map_sources src/map.cpp)
    if(ANDROID)
        list(APPEND _map_sources platform/android/src/main/cpp/map_android.cpp)
    elseif(IOS)
        if(NOT HUXERUI_LIBRARY_GRAPH_ONLY)
            enable_language(OBJCXX)
        endif()
        list(APPEND _map_sources platform/ios/src/map_ios.mm)
        if(HUXERUI_MAP_ENABLE_AMAP)
            list(APPEND _map_sources platform/ios/src/amap_ios.mm)
        endif()
        if(HUXERUI_MAP_ENABLE_BAIDU)
            list(APPEND _map_sources platform/ios/src/baidu_ios.mm)
        endif()
    else()
        list(APPEND _map_sources src/map_unavailable.cpp)
    endif()
    huxerui_add_library(huxerui_map SOURCES ${_map_sources})
else()
    add_library(huxerui_map STATIC ${_map_sources})
endif()
add_library(HuxerUI::Map ALIAS huxerui_map)
if(HUXERUI_LIBRARY_GRAPH_ONLY)
    return()
endif()

target_compile_features(huxerui_map PUBLIC cxx_std_20)
set_target_properties(huxerui_map PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_include_directories(huxerui_map PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>"
    PRIVATE src)
target_compile_options(huxerui_map PRIVATE
    "$<$<CXX_COMPILER_ID:MSVC>:/utf-8;/W4;/permissive->"
    "$<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall;-Wextra;-Wpedantic>")
if(HUXERUI_MAP_BUILD_VIEW)
    target_link_libraries(huxerui_map PUBLIC HuxerUI::huxerui)
    if(IOS)
        include(cmake/iOSMapSDKs.cmake)
    endif()
endif()
if(HUXERUI_MAP_BUILD_TESTS)
    enable_testing()
    add_executable(map_model_tests tests/model_tests.cpp)
    target_link_libraries(map_model_tests PRIVATE HuxerUI::Map)
    add_test(NAME map.model COMMAND map_model_tests)
    if(HUXERUI_MAP_BUILD_VIEW)
        add_executable(map_payload_tests tests/payload_tests.cpp)
        target_include_directories(map_payload_tests PRIVATE src)
        target_link_libraries(map_payload_tests PRIVATE HuxerUI::Map)
        add_test(NAME map.payload COMMAND map_payload_tests)
    endif()
endif()
if(HUXERUI_MAP_BUILD_EXAMPLES)
    if(NOT HUXERUI_MAP_BUILD_VIEW)
        message(FATAL_ERROR "The preview requires HUXERUI_MAP_BUILD_VIEW=ON")
    endif()
    add_subdirectory(examples/preview)
endif()
